Day 3 예시(클래스 기본 형태)

  constructor() {
    this.value = 0; // state
  }
  inc() {
    this.value += 1;
    console.log("Counter.value =", this.value);
  }
}

const c = new Counter();
c.inc();

Day 5 예시(Vue 3 CDN 최소 형태)

수업 당일 CDN 스크립트를 추가해 바로 실행해본다.

<!-- head 또는 body 상단에 추가 -->
<script src="https://unpkg.com/vue@3/dist/vue.global.prod.js"></script>

<div id="app">
  <button @click="count++">증가</button>
  <p>count = {{ count }}</p>
</div>

<script>
  const { createApp } = Vue;
  createApp({
    data() {
      return { count: 0 };
    }
  }).mount("#app");
</script>
